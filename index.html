import os import discord from discord.ext import commands from discord import ui intents = discord.Intents.default() intents.guilds = True intents.members = True intents.guild_messages = True intents.message_content = True bot = commands.Bot(command_prefix="!", intents=intents) class VerifyButton(ui.Button): def __init__(self, role_name: str): super().__init__(label="✅ Verifizieren", style=discord.ButtonStyle.green, custom_id="verify_btn") self.role_name = role_name async def callback(self, interaction: discord.Interaction): role = discord.utils.get(interaction.guild.roles, name=self.role_name) if not role: role = discord.utils.get(interaction.guild.roles, name="Mitglied") if role: try: await interaction.user.add_roles(role) await interaction.response.send_message(f"Erfolgreich verifiziert! Du hast die Rolle **{role.name}** erhalten.", ephemeral=True) except Exception: await interaction.response.send_message("Fehler: Der Bot hat keine Berechtigung, dir diese Rolle zu geben.", ephemeral=True) else: await interaction.response.send_message(f"Die Verifizierungs-Rolle ('{self.role_name}') wurde nicht gefunden!", ephemeral=True) class VerifyView(ui.View): def __init__(self, role_name: str): super().__init__(timeout=None) self.add_item(VerifyButton(role_name)) @bot.command() @commands.has_permissions(administrator=True) async def verify(ctx, channel: discord.TextChannel = None, *, role_name: str = "Mitglied"): target_channel = channel or ctx.channel embed = discord.Embed( title="🔐 Server Verifizierung", description="Klicke auf den Button unten, um dich zu verifizieren!", color=0x3b82f6 ) await target_channel.send(embed=embed, view=VerifyView(role_name)) await ctx.message.delete() @bot.event async def on_ready(): print(f'Eingeloggt als {bot.user.name}') TOKEN = os.getenv("DISCORD_TOKEN") if TOKEN: bot.run(TOKEN) else: print("Fehler: Es wurde keine Umgebungsvariable 'DISCORD_TOKEN' gefunden!")
