import os import discord from discord.ext import commands from discord import ui from aiohttp import web # Intents konfigurieren intents = discord.Intents.default() intents.guilds = True intents.members = True intents.guild_messages = True intents.message_content = True bot = commands.Bot(command_prefix="!", intents=intents) OWNER_ID = "1523728380476919910" # --- HTML / CSS DASHBOARD --- HTML_DASHBOARD = f""" Phantom Admin Center

Phantom Admin Center

ONLINE
Überprüfe Berechtigungen...
Status
✔ Zugriff gewährt (Owner erkannt)

Verbundene Bot-Server

Lade Server...
""" # --- DISCORD UI VERIFIZIERUNG --- class VerifyButton(ui.Button): def __init__(self, role_name: str): super().__init__(label="✅ Verifizieren", style=discord.ButtonStyle.green, custom_id="verify_btn") self.role_name = role_name async def callback(self, interaction: discord.Interaction): role = discord.utils.get(interaction.guild.roles, name=self.role_name) if not role: role = discord.utils.get(interaction.guild.roles, name="Mitglied") if role: try: await interaction.user.add_roles(role) await interaction.response.send_message(f"Erfolgreich verifiziert! Du hast die Rolle **{role.name}** erhalten.", ephemeral=True) except Exception: await interaction.response.send_message("Fehler: Der Bot hat keine Berechtigung, dir diese Rolle zu geben.", ephemeral=True) else: await interaction.response.send_message(f"Die Verifizierungs-Rolle ('{self.role_name}') wurde nicht gefunden!", ephemeral=True) class VerifyView(ui.View): def __init__(self, role_name: str): super().__init__(timeout=None) self.add_item(VerifyButton(role_name)) # --- WEB SERVER ROUTEN --- async def serve_dashboard(request): return web.Response(text=HTML_DASHBOARD, content_type='text/html') async def api_get_bot_servers(request): user_id = request.query.get("userid") if user_id != OWNER_ID: return web.json_response({"error": "Zugriff verweigert."}, status=403) servers_data = [{"id": str(g.id), "name": g.name, "memberCount": g.member_count} for g in bot.guilds] return web.json_response(servers_data) async def start_web_server(): app = web.Application() app.router.add_get('/', serve_dashboard) app.router.add_get('/api/bot-servers', api_get_bot_servers) runner = web.AppRunner(app) await runner.setup() site = web.TCPSite(runner, '0.0.0.0', 8080) await site.start() print("🌐 Webserver erfolgreich gestartet auf http://localhost:8080") # --- BOT EVENTS & BEFEHLE --- @bot.event async def on_ready(): print(f'Eingeloggt als Bot: {bot.user.name}') bot.loop.create_task(start_web_server()) @bot.command() @commands.has_permissions(administrator=True) async def verify(ctx, channel: discord.TextChannel = None, *, role_name: str = "Mitglied"): target_channel = channel or ctx.channel embed = discord.Embed( title="🔐 Server Verifizierung", description="Klicke auf den Button unten, um dich zu verifizieren!", color=0x3b82f6 ) await target_channel.send(embed=embed, view=VerifyView(role_name)) await ctx.message.delete() # --- START --- TOKEN = os.getenv("DISCORD_TOKEN") if TOKEN: bot.run(TOKEN) else: print("Fehler: Es wurde keine Umgebungsvariable 'DISCORD_TOKEN' gefunden!")
