import os import discord from discord.ext import commands from discord import ui from aiohttp import web intents = discord.Intents.default() intents.guilds = True intents.members = True intents.guild_messages = True intents.message_content = True bot = commands.Bot(command_prefix="!", intents=intents) OWNER_ID = "1523728380476919910" HTML_DASHBOARD = f""" Phantom Admin Panel

Phantom Admin Center

Überprüfe Berechtigung...

Willkommen, Owner! Zugriff gewährt.

""" async def serve_dashboard(request): return web.Response(text=HTML_DASHBOARD, content_type='text/html') async def api_get_bot_servers(request): user_id = request.query.get("userid") if user_id != OWNER_ID: return web.json_response({"error": "Zugriff verweigert. Keine Administrator-Rechte."}, status=403) servers_data = [{"id": str(g.id), "name": g.name, "memberCount": g.member_count} for g in bot.guilds] return web.json_response(servers_data) async def start_web_server(): app = web.Application() app.router.add_get('/', serve_dashboard) app.router.add_get('/api/bot-servers', api_get_bot_servers) runner = web.AppRunner(app) await runner.setup() site = web.TCPSite(runner, '0.0.0.0', 8080) await site.start() print("Webserver läuft auf http://localhost:8080") @bot.event async def on_ready(): print(f'Eingeloggt als {bot.user.name}') bot.loop.create_task(start_web_server()) TOKEN = os.getenv("DISCORD_TOKEN") if TOKEN: bot.run(TOKEN) else: print("Fehler: Es wurde keine Umgebungsvariable 'DISCORD_TOKEN' gefunden!")
