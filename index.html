import os import discord from discord.ext import commands from discord import ui from aiohttp import web intents = discord.Intents.default() intents.guilds = True intents.members = True intents.guild_messages = True intents.message_content = True bot = commands.Bot(command_prefix="!", intents=intents) HTML_DASHBOARD = """ Phantom Admin Panel

Phantom Admin Center

Der Webserver läuft stabil!

""" class VerifyButton(ui.Button): def __init__(self, role_name: str): super().__init__(label="Verifizieren", style=discord.ButtonStyle.green, custom_id="verify_btn") self.role_name = role_name async def callback(self, interaction: discord.Interaction): role = discord.utils.get(interaction.guild.roles, name=self.role_name) if role: await interaction.user.add_roles(role) await interaction.response.send_message(f"Rolle {role.name} erhalten!", ephemeral=True) else: await interaction.response.send_message("Rolle nicht gefunden!", ephemeral=True) class VerifyView(ui.View): def __init__(self, role_name: str): super().__init__(timeout=None) self.add_item(VerifyButton(role_name)) async def serve_dashboard(request): return web.Response(text=HTML_DASHBOARD, content_type='text/html') async def api_get_bot_servers(request): servers_data = [{"id": str(g.id), "name": g.name, "memberCount": g.member_count} for g in bot.guilds] return web.json_response(servers_data) async def start_web_server(): app = web.Application() app.router.add_get('/', serve_dashboard) app.router.add_get('/api/bot-servers', api_get_bot_servers) runner = web.AppRunner(app) await runner.setup() site = web.TCPSite(runner, '0.0.0.0', 8080) await site.start() print("Webserver läuft auf http://localhost:8080") @bot.event async def on_ready(): print(f'Eingeloggt als {bot.user.name}') bot.loop.create_task(start_web_server()) # Token wird sicher aus den System-Umgebungsvariablen ausgelesen TOKEN = os.getenv("DISCORD_TOKEN") if TOKEN: bot.run(TOKEN) else: print("Fehler: Es wurde keine Umgebungsvariable 'DISCORD_TOKEN' gefunden!")
